## 重构步骤

### 步骤1: 消除模块加载副作用

**问题：** mysql.py末尾自动调用init_database()，retrieval.py末尾自动connect_milvus()

**修复：** 改为懒初始化单例模式。

```python
# app/core/database.py
import os
import pymysql
from pymysqlpool.pool import Pool

_pool: Pool = None

def get_pool() -> Pool:
    global _pool
    if _pool is None:
        _pool = Pool(
            host=os.getenv("MYSQL_HOST", "127.0.0.1"),
            port=int(os.getenv("MYSQL_PORT", 3306)),
            user=os.getenv("MYSQL_USER", "root"),
            password=os.getenv("MYSQL_PASSWORD"),
            database=os.getenv("MYSQL_DB", "rag_db"),
            charset="utf8mb4",
            size=10,
            max_size=30,
        )
    return _pool

def get_conn():
    return get_pool().get_conn()
```

### 步骤2: 统一LLM调用入口

**问题：** _llm_chat, _llm_chat_stream, _llm_chat_stream_sync, llm_chat.py 四处重复

**修复：** 合并为一个LLMService类。

```python
# app/services/llm_service.py
import os
from openai import AsyncOpenAI, OpenAI
from typing import Optional, AsyncIterator, Iterator

class LLMService:
    def __init__(self):
        self._async_client: Optional[AsyncOpenAI] = None
        self._sync_client: Optional[OpenAI] = None

    @property
    def async_client(self) -> AsyncOpenAI:
        if self._async_client is None:
            self._async_client = AsyncOpenAI(
                api_key=os.getenv("LLM_API_KEY"),
                base_url=os.getenv("LLM_BASE_URL", "https://api.deepseek.com"),
            )
        return self._async_client

    @property
    def sync_client(self) -> OpenAI:
        if self._sync_client is None:
            self._sync_client = OpenAI(
                api_key=os.getenv("LLM_API_KEY"),
                base_url=os.getenv("LLM_BASE_URL", "https://api.deepseek.com"),
            )
        return self._sync_client

    async def chat(self, messages: list, model: str = "deepseek-v4-flash",
                   temperature: float = 0.7, max_tokens: int = 1024) -> str:
        resp = await self.async_client.chat.completions.create(
            model=model, messages=messages,
            temperature=temperature, max_tokens=max_tokens,
        )
        return resp.choices[0].message.content

    async def chat_stream(self, messages: list, model: str = "deepseek-v4-flash",
                          temperature: float = 0.7, max_tokens: int = 1024) -> AsyncIterator[str]:
        stream = await self.async_client.chat.completions.create(
            model=model, messages=messages,
            temperature=temperature, max_tokens=max_tokens, stream=True,
        )
        async for chunk in stream:
            if chunk.choices[0].delta.content:
                yield chunk.choices[0].delta.content

# 全局单例
llm_service = LLMService()
```

### 步骤3: 角色配置外部化

**问题：** role_map={'lawyer':1,'doctor':3,'psych':2} 硬编码

**修复：**

```python
# app/models/role.py
from dataclasses import dataclass
from enum import Enum

class RoleType(str, Enum):
    LAWYER = "lawyer"
    DOCTOR = "doctor"
    PSYCH  = "psych"

@dataclass
class RoleConfig:
    type: RoleType
    db_id: int              # 数据库中的角色ID
    name: str
    system_prompt: str
    temperature: float = 0.7

ROLES: dict[RoleType, RoleConfig] = {
    RoleType.LAWYER: RoleConfig(
        type=RoleType.LAWYER, db_id=1, name="法律顾问",
        system_prompt="你是一位专业的法律顾问...", temperature=0.3,
    ),
    RoleType.PSYCH: RoleConfig(
        type=RoleType.PSYCH, db_id=2, name="心理咨询师",
        system_prompt="你是一位心理咨询师...", temperature=0.7,
    ),
    RoleType.DOCTOR: RoleConfig(
        type=RoleType.DOCTOR, db_id=3, name="医疗顾问",
        system_prompt="你是一位医疗顾问...", temperature=0.3,
    ),
}

def get_role(role_type: str) -> RoleConfig:
    return ROLES[RoleType(role_type)]
```

### 步骤4: 拆分路由层

```python
# app/api/chat.py
from fastapi import APIRouter, Depends
from app.deps import get_current_user, get_chat_service
from app.models.schemas import ChatRequest, ChatResponse

router = APIRouter(prefix="/api/chat", tags=["chat"])

@router.post("", response_model=ChatResponse)
async def chat(
    req: ChatRequest,
    user=Depends(get_current_user),
    svc=Depends(get_chat_service),
):
    return await svc.handle_chat(user.id, req)

@router.get("/stream")
async def chat_stream(
    req: ChatRequest,
    user=Depends(get_current_user),
    svc=Depends(get_chat_service),
):
    return StreamingResponse(
        svc.handle_chat_stream(user.id, req),
        media_type="text/event-stream",
    )
```

### 步骤5: 应用入口精简

```python
# app/main.py
from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from app.config import settings
from app.api import auth, chat, knowledge, admin

app = FastAPI(title="RAG多角色智能问答系统")

app.add_middleware(
    CORSMiddleware,
    allow_origins=settings.cors_origins,
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

app.include_router(auth.router)
app.include_router(chat.router)
app.include_router(knowledge.router)
app.include_router(admin.router)

@app.get("/health")
async def health():
    return {"status": "ok"}
```
